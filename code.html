// --- Hiệu ứng pháo hoa --- const canvasFireworks = document.createElement('canvas'); canvasFireworks.id = 'fireworksCanvas'; canvasFireworks.style.position = 'fixed'; canvasFireworks.style.top = '0'; canvasFireworks.style.left = '0'; canvasFireworks.style.width = '100%'; canvasFireworks.style.height = '100%'; canvasFireworks.style.pointerEvents = 'none'; canvasFireworks.style.zIndex = '9999'; // Đảm bảo hiển thị trên các phần tử khác document.body.appendChild(canvasFireworks); const ctxFireworks = canvasFireworks.getContext('2d'); canvasFireworks.width = window.innerWidth; canvasFireworks.height = window.innerHeight; let fireworks = []; function Spark(x, y, color, size, speedX, speedY) { this.x = x; this.y = y; this.color = color; this.size = size; this.speedX = speedX; this.speedY = speedY; this.life = Math.random() * 60 + 30; // Thời gian tồn tại this.gravity = 0.05; } Spark.prototype.update = function() { this.x += this.speedX; this.y += this.speedY + this.gravity; this.size *= 0.96; this.life--; }; Spark.prototype.draw = function(ctx) { ctx.beginPath(); ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2); ctx.fillStyle = this.color; ctx.fill(); }; function Firework(x, y) { this.x = x; this.y = y; this.sparks = []; const numSparks = Math.floor(Math.random() * 50) + 50; // Số lượng tia lửa const colors = ['#ff5733', '#33ff57', '#5733ff', '#f4ff33', '#ff33e3', '#fff']; for (let i = 0; i < numSparks; i++) { const angle = Math.random() * Math.PI * 2; const speed = Math.random() * 5 + 2; const color = colors[Math.floor(Math.random() * colors.length)]; this.sparks.push(new Spark(x, y, color, Math.random() * 3 + 1, Math.cos(angle) * speed, Math.sin(angle) * speed)); } } Firework.prototype.update = function() { this.sparks.forEach(spark => spark.update()); this.sparks = this.sparks.filter(spark => spark.life > 0); }; Firework.prototype.draw = function(ctx) { this.sparks.forEach(spark => spark.draw(ctx)); }; function animate() { ctxFireworks.fillStyle = 'rgba(0, 0, 0, 0.1)'; // Hiệu ứng mờ dần ctxFireworks.fillRect(0, 0, canvasFireworks.width, canvasFireworks.height); fireworks.forEach((firework, index) => { firework.update(); firework.draw(ctxFireworks); if (firework.sparks.length === 0) { fireworks.splice(index, 1); } }); requestAnimationFrame(animate); } window.addEventListener('resize', () => { canvasFireworks.width = window.innerWidth; canvasFireworks.height = window.innerHeight; }); canvasFireworks.addEventListener('click', (e) => { fireworks.push(new Firework(e.clientX, e.clientY)); }); // Khởi tạo hiệu ứng khi trang tải xong window.addEventListener('load', () => { animate(); // Thêm một vài pháo hoa ban đầu for (let i = 0; i < 3; i++) { setTimeout(() => { fireworks.push(new Firework(Math.random() * window.innerWidth, Math.random() * window.innerHeight / 2)); }, i * 1000); } }); // --- Kết thúc hiệu ứng pháo hoa ---
